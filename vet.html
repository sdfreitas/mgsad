function aplicarGate() { const gate = document.getElementById('authGate'); if (!gate) return; if (siteAutenticado()) { gate.style.display = 'none'; document.body.style.overflow = ''; window.scrollTo(0, 0); } else { gate.style.display = 'flex'; document.body.style.overflow = 'hidden'; const inp = document.getElementById('authInput'); if (inp) setTimeout(() => inp.focus(), 80); } }
